 // Assuming these imports are available in your file
import { Upload } from "@aws-sdk/lib-storage";
// ... other imports for IFile, IResult, Sermon, UploadStatus, ContentStatus, UploadStepType
import BullQueue from "../../queues/queue"; // Your BullQueue instance
import { JobChannel, QueueChannel } from "../../queues/channel.queue"; // Your channel definitions

// NOTE: You must update your BullQueue.addJobs to handle the job structure below.

public async handleUploadSermon(data: IFile): Promise<IResult> {
    let result: IResult = { error: false, message: "", code: 200, data: {} };

    // The stream and metadataStream objects are available here.
    const { stream, metadataStream, info, mimeType, size, fileType, uploadId } = data;

    if (!stream || !metadataStream || !info || !mimeType || !fileType) {
        result.error = true;
        result.code = 400;
        result.message = "Missing required upload fields.";
        return result;
    }

    const folder = await this.getS3Folder(mimeType);
    const s3Key = `${folder}/${uploadId}`;

    try {
        // 1. Upload to S3
        const s3Upload = new Upload({
            client: this.s3Client,
            params: {
                Bucket: this.bucket,
                Key: s3Key,
                Body: stream,
                ContentType: mimeType,
            },
        });

        const s3Response = await s3Upload.done();

        // 2. Prepare and Save upload session in DB
        const uploadSummary = {
            fileId: uploadId,
            fileName: info.filename,
            fileSize: size,
            fileType,
            mimetype: mimeType,
            uploadedBy: data.uploadedBy,
            uploadStatus: UploadStatus.COMPLETED,
            uploadId,
            s3Key: s3Key,
            rawFile: s3Response.Location,
            // metadata will be added by the worker
        };

        const SermonUpload = await Sermon.create({
            uploadSummary,
            status: ContentStatus.PROCESSING, // Use PROCESSING status here
            uploadState: UploadStepType.AUDIO_UPLOADING,
        });

        // 3. ENQUEUE THE METADATA PROCESSING JOB (The new critical step)
        
        // NOTE ON STREAM: Passing a stream in Bull's job data is generally not possible 
        // because the data is serialized (JSON stringified).
        // Since you are using the `metadataStream` **immediately after S3 upload**, 
        // it means you are in the same process and the stream is still active.
        // We will pass the necessary data (the streams and IDs) to the job.
        // If this were a *remote* worker, you'd pass the S3 key and the worker would stream the file from S3.

        const audioJobData = {
            // Using the actual stream objects here works ONLY if the worker is in the same process
            // and consumes the stream before it closes. For robust Bull usage, pass a file/S3 path.
            // However, to satisfy your processor's signature:
            streamForMetadata: metadataStream, 
            mimeType: mimeType,
            uploadId: uploadId, // Used to find and update the Sermon document
        };

        await BullQueue.addJobs({
            queueName: JobChannel.ProcessAudioMetadata, // New queue name
            jobs: [
                {
                    name: QueueChannel.AudioMetadata, // New job name
                    data: audioJobData,
                    options: {
                        jobId: `audio-meta-${uploadId}`, // Unique ID for tracking
                        attempts: 3,
                        delay: 1000,
                    }
                }
            ]
        });

        // 4. Return success immediately
        result.message = "Sermon uploaded and queued for metadata processing.";
        result.data = SermonUpload;
        
        return result;

    } catch (err: any) {
        console.error("Sermon upload failed with a specific error:", err);

        // Ensure both streams are destroyed on failure
        stream.destroy();
        metadataStream.destroy();

        // Cleanup the S3 file if the process failed after upload
        await this.storageService.deleteFile(uploadId as string);
        
        result.error = true;
        result.code = 500;
        result.message = err.message;
        return result;
    }
}
