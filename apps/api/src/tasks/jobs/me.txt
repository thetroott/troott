/**
 * @name audioMetadataProcessor
 * @description The core function that the Bull worker executes for each 'audio:metadata' job.
 * It extracts metadata and updates the database, using the 'done' callback to signal completion.
 * @param job The Bull job object containing the audio data reference
 * @param done The Bull done callback function
 */
const audioMetadataProcessor = async (
    job: Job<IAudioMetadataJobDTO>,
    done: DoneCallback,
) => {
    const { streamForMetadata, mimeType, uploadId } = job.data;
    if (!streamForMetadata || !mimeType || !uploadId) {
        return done(
            new Error(
                'Invalid audio metadata job payload: streamForMetadata, mimeType, and uploadId are required.',
            ),
        );
    }

    try {
        let metadata: IAudioMetadata;

        const { parseStream } = await import('music-metadata');

        const mmMetadata = await parseStream(streamForMetadata, mimeType, {
            duration: true,
        });

        metadata = {
            metadataType: FileType.AUDIO,
            formatName: mmMetadata.format.container,
            codec: mmMetadata.format.codec,
            duration: mmMetadata.format.duration,
            bitrate: mmMetadata.format.bitrate,
            year: mmMetadata.common.year,
        };

        const query = sermonId
            ? { _id: sermonId }
            : { 'uploadSummary.uploadId': uploadId };
        const updateSermon = await Sermon.findOneAndUpdate(
            query,
            {
                $set: {
                    'uploadSummary.metadata': metadata,
                    status: 'DRAFT',
                },
            },
            { new: true },
        );

        if (!updateSermon) {
            const errorMessage = `Sermon with uploadId ${uploadId} not found.`;
            logger.log({
                data: errorMessage,
                label: 'audio-metadata-processor',
                type: 'error',
            });
            return done(new Error(errorMessage));
        }

        logger.log({
            data: `Metadata extracted and Sermon updated successfully for job ID: ${job.id}`,
            label: 'audio-metadata-processor',
            type: 'success',
        });

        done(null, updateSermon);
    } catch (error) {
        logger.log({
            data: `Critical error during processing of Audio Job ID: ${job.id}. Error: ${error instanceof Error ? error.message : String(error)}`,
            label: 'audio-processor-critical',
            type: 'error',
        });

        done(error as Error);
    }
};

export default audioMetadataProcessor;

const audioHLSProcessor = async (
  job: Job<IAudioHLSJobDTO>,
  done: DoneCallback
) => {
  const { uploadId, renditions, inputStream } = job.data;

  try {
    // Prepare output streams for each rendition
    const outputStreams: Record<string, PassThrough> = {};
    const renditionsH: AudioRenditionDTO[] = renditions?.length
      ? renditions
      : [
          { name: "low", bitrate: 64, sampleRate: 44100, channels: 2 },
          { name: "medium", bitrate: 128, sampleRate: 44100, channels: 2 },
          { name: "high", bitrate: 192, sampleRate: 44100, channels: 2 },
        ];

    for (const r of renditionsH) {
      outputStreams[r.name] = new PassThrough();
    }

    // Process HLS fully in-memory
    const result = await AudioProcessing.ProcessHLS({
      inputStream,
      outputStreams,
      renditions: renditionsH,
      segmentDuration: 6,
    });

    // Upload each rendition to storage
    const uploadedFiles = [];
    for (const r of renditionsH) {
      const uploadResult = await storageService.uploadFile({
        stream: outputStreams[r.name],
        uploadId: `${uploadId}-${r.name}`,
        size: 0, // streaming, unknown size
      });
      uploadedFiles.push(uploadResult.data);
    }

    logger.log({
      data: `HLS packaged for job ID ${job.id}`,
      label: "audio-hls-processor",
      type: "success",
    });

    done(null, uploadedFiles);
  } catch (err: any) {
    logger.log({
      data: `HLS packaging failed for job ID ${job.id}. Error: ${err.message}`,
      label: "audio-hls-processor",
      type: "error",
    });
    done(err);
  }

export class AudioProcessing {
    public async ProcessDASH(data: DASHDTO): Promise<IResult> {
        const { inputStream, renditions, outputStreams, segmentDuration = 6 } = data;
        const result: IResult = { error: false, message: "", code: 200, data: {} };

        try {
            // We only handle single audio stream per rendition for simplicity
            for (const rendition of renditions) {
                const ffArgs: string[] = [
                    "-i", "pipe:0",
                    "-map", "0:a",
                    "-c:a", "aac",
                    "-b:a", `${rendition.bitrate}k`,
                    "-f", "segment",
                    "-segment_time", `${segmentDuration}`,
                    "-reset_timestamps", "1",
                    "pipe:1" // each segment goes to stdout
                ];

                const segmentStream = new PassThrough();
                outputStreams[rendition.name] = segmentStream;

                await this.spawnFFmpeg({
                    args: ffArgs,
                    inputStream,
                    outputStream: segmentStream
                });
            }

            result.data = { outputStreams };
            return result;
        } catch (err: any) {
            result.error = true;
            result.code = 500;
            result.message = err.message;
            return result;
        }
    }

    private async spawnFFmpeg(data: { args: string[], inputStream: PassThrough, outputStream?: PassThrough }): Promise<IResult> {
        const { args, inputStream, outputStream } = data;
        const result: IResult = { error: false, message: "", code: 200, data: {} };

        try {
            await new Promise<void>((resolve, reject) => {
                const ff = spawn("ffmpeg", args);

                ff.stdin.on("error", reject);
                inputStream.pipe(ff.stdin);

                if (outputStream) {
                    ff.stdout.pipe(outputStream);
                } else {
                    ff.stdout.resume();
                }

                ff.stderr.on("data", chunk => {
                    // optional: handle FFmpeg logs
                });

                ff.on("close", code => {
                    if (code === 0) resolve();
                    else reject(new Error("FFmpeg exited with code " + code));
                });

                ff.on("error", reject);
            });

            return result;
        } catch (err: any) {
            result.error = true;
            result.code = 500;
            result.message = err.message;
            return result;
        }
    }
}


import { spawn } from "child_process";
import { PassThrough } from "stream";
import { IResult } from "../../utils/interfaces.util";
import { DASHDTO, FFmpegOptionsDTO } from "../../dtos/audio.dto";

class AudioProcessing {

    public async ProcessDASH(data: DASHDTO): Promise<IResult> {
        const { inputStream, renditions, segmentDuration = 6 } = data;
        const result: IResult = { error: false, message: "", code: 200, data: {} };

        try {
            // Streams to hold manifest and segments in memory
            const manifestStream = new PassThrough();
            const segments = new Map<string, PassThrough>();

            // FFmpeg args
            const args: string[] = [
                "-i", "pipe:0",
                "-map", "0:a",
                "-c:a", "aac",
                "-b:a", `${renditions[0].bitrate}k`,
                "-use_template", "1",
                "-use_timeline", "1",
                "-seg_duration", `${segmentDuration}`,
                "-adaptation_sets", "id=0,streams=a",
                "-f", "dash",
                "pipe:1" // manifest will go to stdout
            ];

            // Collect manifest chunks
            const manifestChunks: Buffer[] = [];
            manifestStream.on("data", chunk => manifestChunks.push(Buffer.from(chunk)));

            await this.spawnFFmpeg({ args, inputStream, outputStream: manifestStream });

            // Return manifest in memory
            result.data = {
                manifest: Buffer.concat(manifestChunks),
                segments // segments Map can be populated if you stream segments individually
            };
            return result;

        } catch (err: any) {
            result.error = true;
            result.code = 500;
            result.message = err.message;
            return result;
        }
    }

    private async spawnFFmpeg(data: FFmpegOptionsDTO): Promise<IResult> {
        const { args, inputStream, outputStream, onData } = data;
        const result: IResult = { error: false, message: "", code: 200, data: {} };

        try {
            await new Promise<void>((resolve, reject) => {
                const ff = spawn("ffmpeg", args);

                ff.stdin.on("error", reject);
                inputStream.pipe(ff.stdin);

                if (outputStream) {
                    ff.stdout.pipe(outputStream);
                } else {
                    ff.stdout.resume();
                }

                ff.stderr.on("data", chunk => {
                    if (onData) onData.push(chunk.toString());
                });

                ff.on("close", code => {
                    if (code === 0) resolve();
                    else reject(new Error("FFmpeg exited with code " + code));
                });

                ff.on("error", reject);
            });

            return result;
        } catch (err: any) {
            result.error = true;
            result.code = 500;
            result.message = err.message;
            return result;
        }
    }
}

function dtoToArgs(opts: FFmpegOptions): string[] {
  const args: string[] = [];

  if (opts.input) args.push("-i", opts.input);
  if (opts.codec) args.push("-c:a", opts.codec);
  if (opts.bitrate) args.push("-b:a", opts.bitrate);
  if (opts.format) args.push("-f", opts.format);
  if (opts.output) args.push(opts.output);

  return args;
}

    return {
      integrated: this.extractNumber(text, "I:"),
      loudnessRange: this.extractNumber(text, "LRA:"),
      truePeak: this.extractNumber(text, "TP:")
    };
  }

  private extractNumber(text: string, key: string): number {
    const match = text.match(new RegExp(`${key}\\s*([-0-9.]+)`));
    return match ? parseFloat(match[1]) : 0;
  }
}

import { Job, DoneCallback } from "bull";
import { PassThrough } from "stream";

import logger from "../../utils/logger.util";

const audioHLSProcessor = async (
  job: Job<IAudioHLSJobDTO>,
  done: DoneCallback
) => {
  const { uploadId, renditions, inputStream, mimeType } = job.data;

  try {
    if (!(inputStream instanceof PassThrough)) {
      throw new Error("inputStream must be a PassThrough stream");
    }

    // Default renditions if none provided
    const renditionsH: AudioRenditionDTO[] = renditions?.length
      ? renditions
      : [

          { name: "medium", bitrate: 128, sampleRate: 44100, channels: 2 },
          { name: "high", bitrate: 192, sampleRate: 44100, channels: 2 },
        ];

    // FFmpeg outputs a PassThrough per rendition
    const renditionStreams: Record<string, PassThrough> = {};
    for (const r of renditionsH) {
      renditionStreams[r.name] = new PassThrough();
    }

    // Process HLS entirely in-memory
    const result = await AudioProcessing.ProcessHLS({

      renditions: renditionsH,
      segmentDuration: 6,

    });

    // Upload each rendition to S3
    const uploadedFiles = [];
    for (const [name, stream] of Object.entries(renditionStreams)) {
      const uploadResult = await storageService.uploadFile({
        stream,
        mimeType,
        uploadId: `${uploadId}-${name}`,
        info: { filename: `${name}.m3u8` },
        size: 0, // streaming
        fileType: "audio",
      });
      uploadedFiles.push(uploadResult.data);
    }

    logger.log({
      data: `HLS packaged and uploaded for Job ID ${job.id}`,
      label: "audio-hls-processor",
      type: "success",
    });

    done(null, uploadedFiles);
  } catch (err: any) {
    logger.log({
      data: `HLS packaging failed for Job ID ${job.id}. Error: ${err.message}`,
      label: "audio-hls-processor",
      type: "error",
    });
    done(err);
  }
};

export default audioHLSProcessor;

public async ProcessHLS(data: HLSDTO & { outputStreams?: Record<string, PassThrough> }): Promise<IResult> {
  const { inputStream, renditions, segmentDuration = 6, outputStreams } = data;
  const result: IResult = { error: false, message: "", code: 200, data: {} };

  try {
    const outputs: { name: string }[] = [];

    for (const preset of renditions) {
      const args = [
        "-i", "pipe:0",
        "-c:a", "aac",
        "-b:a", `${preset.bitrate}k`,
        "-ar", `${preset.sampleRate}`,
        "-ac", `${preset.channels}`,
        "-f", "hls",
        "-hls_time", `${segmentDuration}`,
        "-hls_playlist_type", "vod",
        "pipe:1" // output to stdout stream
      ];

      const outStream = outputStreams?.[preset.name] || new PassThrough();
      await this.spawnFFmpeg({ args, inputStream, outputStream: outStream });

      outputs.push({ name: preset.name });
    }

    result.data = outputs;
    return result;
  } catch (err: any) {
    result.error = true;
    result.code = 500;
    result.message = err.message;
    return result;
  }
}
